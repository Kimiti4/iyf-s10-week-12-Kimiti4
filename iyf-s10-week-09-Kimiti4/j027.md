```javascript
/**
 * Requirements Compiler
 * Transforms product specification into normalized requirements
 */

const fs = require('fs');
const path = require('path');

class RequirementsCompiler {
  constructor(spec) {
    this.spec = spec;
    this.requirements = [];
    this.diagnostics = [];
  }

  compile() {
    console.log('📋 Compiling requirements...');
    
    this.extractRequirements();
    this.validateRequirements();
    this.detectConflicts();
    this.generateDiagnostics();

    return {
      requirements: this.requirements,
      diagnostics: this.diagnostics,
      summary: this.generateSummary()
    };
  }

  extractRequirements() {
    const features = this.spec.features || [];
    
    features.forEach(feature => {
      feature.requirements.forEach(req => {
        const requirement = {
          id: req.id,
          feature_id: feature.id,
          feature_name: feature.name,
          description: req.description,
          priority: req.priority || 'medium',
          acceptance_criteria: req.acceptance_criteria || [],
          traceability: {
            workflows: this.findRelatedWorkflows(req),
            entities: this.findRelatedEntities(req),
            api_endpoints: this.findRelatedEndpoints(req)
          }
        };

        this.requirements.push(requirement);
      });
    });
  }

  findRelatedWorkflows(requirement) {
    const workflows = this.spec.workflows || [];
    return workflows
      .filter(wf => 
        wf.description.toLowerCase().includes(requirement.description.toLowerCase()) ||
        wf.steps.some(step => 
          step.description.toLowerCase().includes(requirement.description.toLowerCase())
        )
      )
      .map(wf => wf.id);
  }

  findRelatedEntities(requirement) {
    const entities = this.spec.entities || [];
    const description = requirement.description.toLowerCase();
    
    return entities
      .filter(entity => description.includes(entity.name.toLowerCase()))
      .map(entity => entity.id);
  }

  findRelatedEndpoints(requirement) {
    const endpoints = this.spec.api_requirements || [];
    const description = requirement.description.toLowerCase();
    
    return endpoints
      .filter(endpoint => 
        description.includes(endpoint.description.toLowerCase()) ||
        description.includes(endpoint.endpoint.toLowerCase())
      )
      .map(endpoint => endpoint.endpoint);
  }

  validateRequirements() {
    const ids = new Set();
    
    this.requirements.forEach(req => {
      // Check for duplicate IDs
      if (ids.has(req.id)) {
        this.diagnostics.push({
          type: 'ERROR',
          code: 'DUPLICATE_REQ_ID',
          message: `Duplicate requirement ID: ${req.id}`,
          requirement: req.id
        });
      }
      ids.add(req.id);

      // Check for missing acceptance criteria
      if (req.acceptance_criteria.length === 0) {
        this.diagnostics.push({
          type: 'WARNING',
          code: 'MISSING_ACCEPTANCE_CRITERIA',
          message: `Requirement ${req.id} has no acceptance criteria`,
          requirement: req.id
        });
      }

      // Check for orphaned requirements (no traceability)
      if (
        req.traceability.workflows.length === 0 &&
        req.traceability.entities.length === 0 &&
        req.traceability.api_endpoints.length === 0
      ) {
        this.diagnostics.push({
          type: 'WARNING',
          code: 'ORPHANED_REQUIREMENT',
          message: `Requirement ${req.id} has no traceable artifacts`,
          requirement: req.id
        });
      }
    });
  }

  detectConflicts() {
    // Check for contradictory requirements
    const descriptions = this.requirements.map(r => r.description.toLowerCase());
    
    this.requirements.forEach(req => {
      const desc = req.description.toLowerCase();
      
      // Check for negation conflicts
      if (desc.includes('not') || desc.includes('never')) {
        const positive = descriptions.find(d => 
          d !== desc && 
          d.replace('not', '').replace('never', '').includes(desc.replace('not', '').replace('never', ''))
        );
        
        if (positive) {
          this.diagnostics.push({
            type: 'WARNING',
            code: 'POTENTIAL_CONFLICT',
            message: `Requirement ${req.id} may conflict with another requirement`,
            requirement: req.id
          });
        }
      }
    });
  }

  generateDiagnostics() {
    const errors = this.diagnostics.filter(d => d.type === 'ERROR');
    const warnings = this.diagnostics.filter(d => d.type === 'WARNING');

    console.log(`   Found ${errors.length} errors, ${warnings.length} warnings`);

    if (errors.length > 0) {
      console.error('\n❌ Requirements compilation errors:');
      errors.forEach(err => console.error(`   ${err.code}: ${err.message}`));
      throw new Error(`Requirements compilation failed with ${errors.length} errors`);
    }

    if (warnings.length > 0) {
      console.warn('\n⚠️  Requirements compilation warnings:');
      warnings.forEach(warn => console.warn(`   ${warn.code}: ${warn.message}`));
    }
  }

  generateSummary() {
    const byPriority = {
      critical: this.requirements.filter(r => r.priority === 'critical').length,
      high: this.requirements.filter(r => r.priority === 'high').length,
      medium: this.requirements.filter(r => r.priority === 'medium').length,
      low: this.requirements.filter(r => r.priority === 'low').length
    };

    return {
      total: this.requirements.length,
      byPriority,
      withAcceptanceCriteria: this.requirements.filter(r => r.acceptance_criteria.length > 0).length,
      withTraceability: this.requirements.filter(r => 
        r.traceability.workflows.length > 0 ||
        r.traceability.entities.length > 0 ||
        r.traceability.api_endpoints.length > 0
      ).length
    };
  }
}

module.exports = RequirementsCompiler;
```

```javascript
/**
 * Domain Model Compiler
 * Transforms entities into normalized domain model
 */

class DomainCompiler {
  constructor(spec) {
    this.spec = spec;
    this.entities = [];
    this.relationships = [];
    this.diagnostics = [];
  }

  compile() {
    console.log('🏗️  Compiling domain model...');
    
    this.extractEntities();
    this.extractRelationships();
    this.validateDomain();
    this.detectCircularDependencies();

    return {
      entities: this.entities,
      relationships: this.relationships,
      diagnostics: this.diagnostics,
      summary: this.generateSummary()
    };
  }

  extractEntities() {
    const entities = this.spec.entities || [];
    
    entities.forEach(entity => {
      const compiledEntity = {
        id: entity.id,
        name: entity.name,
        description: entity.description,
        table_name: this.toTableName(entity.name),
        fields: this.compileFields(entity.fields),
        indexes: this.generateIndexes(entity.fields),
        validation_rules: this.generateValidationRules(entity.fields)
      };

      this.entities.push(compiledEntity);
    });
  }

  compileFields(fields) {
    return fields.map(field => ({
      name: field.name,
      column_name: field.name,
      type: this.mapType(field.type),
      database_type: this.getDatabaseType(field.type, field),
      required: field.required || false,
      unique: field.unique || false,
      indexed: field.indexed || false,
      default_value: field.default,
      validation: field.validation || {},
      enum_values: field.enum_values || null
    }));
  }

  mapType(specType) {
    const typeMap = {
      'string': 'string',
      'number': 'number',
      'boolean': 'boolean',
      'date': 'date',
      'datetime': 'datetime',
      'uuid': 'uuid',
      'enum': 'enum',
      'array': 'array',
      'object': 'object'
    };
    return typeMap[specType] || 'string';
  }

  getDatabaseType(specType, field) {
    const dbTypeMap = {
      'string': field.validation?.maxLength ? `VARCHAR(${field.validation.maxLength})` : 'TEXT',
      'number': 'DECIMAL(10,2)',
      'boolean': 'BOOLEAN',
      'date': 'DATE',
      'datetime': 'TIMESTAMP',
      'uuid': 'UUID',
      'enum': `VARCHAR(50)` // Will add CHECK constraint
    };
    return dbTypeMap[specType] || 'TEXT';
  }

  toTableName(entityName) {
    return entityName.toLowerCase().replace(/-/g, '_') + 's';
  }

  generateIndexes(fields) {
    return fields
      .filter(f => f.indexed || f.unique)
      .map(f => ({
        field: f.name,
        unique: f.unique,
        index_name: `idx_${f.name}`
      }));
  }

  generateValidationRules(fields) {
    return fields
      .filter(f => f.validation && Object.keys(f.validation).length > 0)
      .map(f => ({
        field: f.name,
        rules: f.validation
      }));
  }

  extractRelationships() {
    this.spec.entities.forEach(entity => {
      const relationships = entity.relationships || [];
      
      relationships.forEach(rel => {
        this.relationships.push({
          source: entity.id,
          target: rel.target,
          type: rel.type,
          onDelete: rel.onDelete || 'restrict',
          foreign_key: this.generateForeignKey(entity, rel)
        });
      });
    });
  }

  generateForeignKey(entity, relationship) {
    const targetEntity = this.spec.entities.find(e => e.id === relationship.target);
    if (!targetEntity) return null;

    return {
      column: `${targetEntity.name.toLowerCase()}_id`,
      references_table: this.toTableName(targetEntity.name),
      references_column: 'id'
    };
  }

  validateDomain() {
    const entityIds = new Set(this.entities.map(e => e.id));
    
    // Check for missing relationship targets
    this.relationships.forEach(rel => {
      if (!entityIds.has(rel.target)) {
        this.diagnostics.push({
          type: 'ERROR',
          code: 'INVALID_RELATIONSHIP_TARGET',
          message: `Relationship from ${rel.source} references non-existent entity ${rel.target}`
        });
      }
    });

    // Check for duplicate field names within entities
    this.entities.forEach(entity => {
      const fieldNames = new Set();
      entity.fields.forEach(field => {
        if (fieldNames.has(field.name)) {
          this.diagnostics.push({
            type: 'ERROR',
            code: 'DUPLICATE_FIELD_NAME',
            message: `Entity ${entity.name} has duplicate field: ${field.name}`
          });
        }
        fieldNames.add(field.name);
      });
    });
  }

  detectCircularDependencies() {
    // Simple cycle detection in relationships
    const graph = {};
    this.relationships.forEach(rel => {
      if (!graph[rel.source]) graph[rel.source] = [];
      graph[rel.source].push(rel.target);
    });

    const visited = new Set();
    const recursionStack = new Set();

    const hasCycle = (node) => {
      visited.add(node);
      recursionStack.add(node);

      const neighbors = graph[node] || [];
      for (const neighbor of neighbors) {
        if (!visited.has(neighbor)) {
          if (hasCycle(neighbor)) return true;
        } else if (recursionStack.has(neighbor)) {
          return true;
        }
      }

      recursionStack.delete(node);
      return false;
    };

    for (const node of Object.keys(graph)) {
      if (!visited.has(node)) {
        if (hasCycle(node)) {
          this.diagnostics.push({
            type: 'WARNING',
            code: 'CIRCULAR_DEPENDENCY',
            message: 'Circular dependency detected in entity relationships'
          });
          break;
        }
      }
    }
  }

  generateSummary() {
    return {
      entities: this.entities.length,
      relationships: this.relationships.length,
      fields: this.entities.reduce((sum, e) => sum + e.fields.length, 0),
      indexes: this.entities.reduce((sum, e) => sum + e.indexes.length, 0)
    };
  }
}

module.exports = DomainCompiler;
```

```javascript
  generateModelCode(entity) {
    const fields = entity.fields
      .filter(f => f.name !== 'id')
      .map(f => `    ${f.name}: data.${f.name}`)
      .join(',\n');

    return `const { pool } = require('../database/connection');

class ${entity.name} {
  static async create(data) {
    const query = \`
      INSERT INTO ${entity.table_name} (${entity.fields.filter(f => f.name !== 'id').map(f => f.name).join(', ')})
      VALUES (${entity.fields.filter(f => f.name !== 'id').map((_, i) => `$${i + 1}`).join(', ')})
      RETURNING *
    \`;
    
    const values = [${entity.fields.filter(f => f.name !== 'id').map(f => `data.${f.name}`).join(', ')}];
    const result = await pool.query(query, values);
    return result.rows[0];
  }

  static async findById(id) {
    const query = 'SELECT * FROM ${entity.table_name} WHERE id = $1';
    const result = await pool.query(query, [id]);
    return result.rows[0];
  }

  static async findAll(filters = {}) {
    let query = 'SELECT * FROM ${entity.table_name}';
    const values = [];
    const conditions = [];

    Object.keys(filters).forEach((key, index) => {
      conditions.push(\`\${key} = $\${index + 1}\`);
      values.push(filters[key]);
    });

    if (conditions.length > 0) {
      query += ' WHERE ' + conditions.join(' AND ');
    }

    query += ' ORDER BY created_at DESC';
    const result = await pool.query(query, values);
    return result.rows;
  }

  static async update(id, data) {
    const fields = Object.keys(data);
    const values = Object.values(data);
    
    const setClause = fields.map((field, index) => \`\${field} = $\${index + 1}\`).join(', ');
    const query = \`
      UPDATE ${entity.table_name}
      SET \${setClause}, updated_at = NOW()
      WHERE id = $\${fields.length + 1}
      RETURNING *
    \`;
    
    values.push(id);
    const result = await pool.query(query, values);
    return result.rows[0];
  }

  static async delete(id) {
    const query = 'DELETE FROM ${entity.table_name} WHERE id = $1';
    await pool.query(query, [id]);
  }
}

module.exports = ${entity.name};
`;
  }

  generateRoutes() {
    const routes = this.spec.routes || [];
    const apiRoutes = this.spec.api_requirements || [];

    apiRoutes.forEach(route => {

  generateRouteCode(resource, endpoints) {
    const controllerName = resource.charAt(0).toUpperCase() + resource.slice(1) + 'Controller';
    
    const routeDefinitions = endpoints.map(endpoint => {
      const method = endpoint.method.toLowerCase();
      const path = endpoint.endpoint.replace(`/api/${resource}`, '') || '/';
      const auth = endpoint.authentication ? ', requireAuth' : '';
      const roles = endpoint.authorization?.length ? `, requireRole(${JSON.stringify(endpoint.authorization)})` : '';
      
      let handler = '';
      if (method === 'get' && path === '/') handler = 'getAll';
      else if (method === 'get' && path.includes(':id')) handler = 'getById';
      else if (method === 'post' && path === '/') handler = 'create';
      else if (method === 'put' && path.includes(':id')) handler = 'update';
      else if (method === 'delete' && path.includes(':id')) handler = 'delete';
      else handler = 'getAll'; // Fallback

      return `router.${method}('${path}'${auth}${roles}, ${controllerName}.${handler});`;
    }).join('\n');

    return `const express = require('express');
const router = express.Router();
const ${controllerName} = require('../controllers/${controllerName}');
const { requireAuth, requireRole } = require('../middleware/auth');

${routeDefinitions}

module.exports = router;
`;
  }

  generateMiddleware() {

const requireAuth = (req, res, next) => {
  const token = req.headers.authorization?.split(' ')[1];
  
  if (!token) {
    return res.status(401).json({ success: false, error: 'Authentication required' });

    return res.status(401).json({ success: false, error: 'Invalid token' });

const requireRole = (roles) => {

};

module.exports = { requireAuth, requireRole };
`;

  generateMigrations() {
    const migrations = [];
    
    // Enable UUID extension
    migrations.push('CREATE EXTENSION IF NOT EXISTS "uuid-ossp";');
    migrations.push('');

    // Create tables
    this.domain.entities.forEach(entity => {
      const fields = entity.fields.map(f => {
        let definition = `  ${f.column_name} ${f.database_type}`;
        if (f.required) definition += ' NOT NULL';
        if (f.unique) definition += ' UNIQUE';
        if (f.default_value) definition += ` DEFAULT ${f.default_value}`;
        return definition;
      }).join(',\n');

      const foreignKeys = entity.fields
        .filter(f => f.name.endsWith('_id'))
        .map(f => {
          const refTable = f.name.replace('_id', '') + 's';
          return `  FOREIGN KEY (${f.column_name}) REFERENCES ${refTable}(id) ON DELETE CASCADE`;
        })
        .join(',\n');

      let tableSQL = `CREATE TABLE IF NOT EXISTS ${entity.table_name} (\n${fields}`;
      if (foreignKeys) tableSQL += ',\n' + foreignKeys;
      tableSQL += '\n);';

      migrations.push(tableSQL);
      migrations.push('');

      // Indexes
      entity.indexes.forEach(index => {
        const unique = index.unique ? 'UNIQUE ' : '';
        migrations.push(`CREATE ${unique}INDEX IF NOT EXISTS ${index.index_name} ON ${entity.table_name}(${index.field});`);
      });
    });

    return migrations.join('\n');
  }
```

```javascript
function loadSpecification(specPath) {
  const fullPath = path.resolve(specPath);

  const spec = JSON.parse(content);
```
